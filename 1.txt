# Practical 1A – Add Two Numbers and Store in Blocks (JavaScript)

## 1. Code

**File: add.js**
```javascript
const crypto = require('crypto');

class Block {
  constructor(index, timestamp, n1, n2, previousHash = '') {
    this.index = index;
    this.timestamp = timestamp;
    this.n1 = n1;
    this.n2 = n2;
    this.sum = n1 + n2;
    this.previousHash = previousHash;
    this.hash = this.calculateHash();
  }

  calculateHash() {
    return crypto.createHash('sha256')
      .update(`${this.index}${this.timestamp}${this.sum}${this.previousHash}`)
      .digest('hex');
  }
}

class Blockchain {
  constructor() {
    this.chain = [new Block(0, Date.now(), 0, 0, '0')];
  }

  getLatestBlock() { 
    return this.chain[this.chain.length - 1]; 
  }

  addBlock(block) {
    block.previousHash = this.getLatestBlock().hash;
    block.hash = block.calculateHash();
    this.chain.push(block);
  }
}

module.exports = { Block, Blockchain };
```

**File: test.js**
```javascript
const { Block, Blockchain } = require('./add');
const chain = new Blockchain();

chain.addBlock(new Block(1, Date.now(), 12, 24));
chain.addBlock(new Block(2, Date.now(), 58, 9));

console.log(JSON.stringify(chain, null, 2));
```

# Practical 1B – Prime Number Check Using Blocks (JavaScript)

## 1. Code

**File: prime.js**

```javascript
const crypto = require('crypto');

class Block {
  constructor(index, timestamp, number, previousHash = '') {
    this.index = index;
    this.timestamp = timestamp;
    this.number = number;
    this.result = this.isPrime(number);
    this.previousHash = previousHash;
    this.hash = this.calculateHash();
  }

  isPrime(n) {
    if (n < 2) return false;
    for (let i = 2; i <= Math.sqrt(n); i++)
      if (n % i === 0) return false;
    return true;
  }

  calculateHash() {
    return crypto.createHash('sha256')
      .update(`${this.index}${this.timestamp}${this.number}${this.result}${this.previousHash}`)
      .digest('hex');
  }
}

class Blockchain {
  constructor() { 
    this.chain = [new Block(0, Date.now(), 0, '0')]; 
  }
  
  addBlock(block) {
    block.previousHash = this.chain[this.chain.length - 1].hash;
    block.hash = block.calculateHash();
    this.chain.push(block);
  }
}

module.exports = { Block, Blockchain };
```

# Practical 1C – KrishnaMurthy Number Check Using Blocks

## 1. Code

**File: krishnamurthy.js**
```javascript
const crypto = require('crypto');
 
class Block {
  constructor(index, number, previousHash = '0') {
    this.index = index;
    this.timestamp = Date.now();
    this.number = number;
    this.result = this.isKrishnaMurthy(number);
    this.previousHash = previousHash;
    this.hash = this.calculateHash();
  }
 
  factorial(n) {
    let fact = 1;
    for (let i = 2; i <= n; i++) fact *= i;
    return fact;
  }
 
  isKrishnaMurthy(n) {
    if (n < 0) return false;
    let temp = n, sum = 0;
    do {
      sum += this.factorial(temp % 10);
      temp = Math.floor(temp / 10);
    } while (temp > 0);
    return sum === n;
  }
 
  calculateHash() {
    return crypto.createHash('sha256')
      .update(`${this.index}${this.timestamp}${this.number}${this.result}${this.previousHash}`)
      .digest('hex');
  }
}
 
class Blockchain {
  constructor() { this.chain = [new Block(0, 0)]; }
  addNumber(n) {
    const previous = this.chain[this.chain.length - 1];
    this.chain.push(new Block(this.chain.length, n, previous.hash));
  }
}
 
const chain = new Blockchain();
```

# Practical 2C – Kaprekar Number Check Using Blocks (JavaScript)

## 1. Code

**File: kaprekar.js**
```javascript
const crypto = require('crypto');

class Block {
  constructor(index, number, previousHash = '0') {
    this.index = index;
    this.timestamp = Date.now();
    this.number = number;
    this.result = this.isKaprekar(number);
    this.previousHash = previousHash;
    this.hash = this.calculateHash();
  }

  isKaprekar(n) {
    if (n === 1) return true;
    if (n <= 0) return false;
    const square = String(n * n);
    for (let i = 1; i < square.length; i++) {
      const left = Number(square.slice(0, i));
      const right = Number(square.slice(i));
      if (left + right === n) return true;
    }
    return false;
  }

  calculateHash() {
    return crypto.createHash('sha256')
      .update(`${this.index}${this.timestamp}${this.number}${this.result}${this.previousHash}`)
      .digest('hex');
  }
}

class Blockchain {
  constructor() { this.chain = [new Block(0, 1)]; }
  addNumber(n) {
    const prev = this.chain[this.chain.length - 1];
    this.chain.push(new Block(this.chain.length, n, prev.hash));
  }
}
```

# Practical 3A – Anagram Checker Blockchain (Python)

## 1. Code

**File: anagrams.py**
```python
import hashlib
from datetime import datetime

class Block:
    def __init__(self, index, word1, word2, result, previous_hash):
        self.index = index
        self.timestamp = str(datetime.now())
        self.word1 = word1
        self.word2 = word2
        self.result = result
        self.previous_hash = previous_hash
        self.hash = self.calculate_hash()

    def calculate_hash(self):
        data = f"{self.index}{self.timestamp}{self.word1}{self.word2}{self.result}{self.previous_hash}"
        return hashlib.sha256(data.encode()).hexdigest()

class Blockchain:
    def __init__(self):
        self.chain = [Block(0, "Genesis", "Genesis", True, "0")]

    def add_block(self, word1, word2):
        result = sorted(word1.lower()) == sorted(word2.lower())
        self.chain.append(Block(len(self.chain), word1, word2, result, self.chain[-1].hash))

    def is_valid(self):
        for i in range(1, len(self.chain)):
            current, previous = self.chain[i], self.chain[i - 1]
            if current.hash != current.calculate_hash() or current.previous_hash != previous.hash:
                return False
        return True

chain = Blockchain()
chain.add_block("bread", "apple")
chain.add_block("listen", "silent")

for b in chain.chain:
    print(b.index, b.word1, b.word2, b.result, b.hash)
    
print("Blockchain valid:", chain.is_valid())
```

## 1. Code

**File: transactions.py**

```python
import hashlib
from datetime import datetime

class Block:
    def __init__(self, index, operation, balance, previous_hash):
        self.index = index
        self.timestamp = str(datetime.now())
        self.operation = operation
        self.balance = balance
        self.previous_hash = previous_hash
        self.hash = self.calculate_hash()

    def calculate_hash(self):
        data = f"{self.index}{self.timestamp}{self.operation}{self.balance}{self.previous_hash}"
        return hashlib.sha256(data.encode()).hexdigest()

class Blockchain:
    def __init__(self):
        self.chain = [Block(0, "d", 1000, "0")]

    def add_transaction(self, operation, amount):
        if amount <= 0:
            print("Amount must be positive")
            return
            
        balance = self.chain[-1].balance
        operation = operation.lower()
        
        if operation == "d":
            balance += amount
        elif operation == "w" and amount <= balance:
            balance -= amount
        elif operation == "w":
            print("Insufficient balance")
            return
        else:
            print("Use d for deposit or w for withdrawal")
            return
            
        self.chain.append(Block(len(self.chain), operation, balance, self.chain[-1].hash))

    def is_valid(self):
        for i in range(1, len(self.chain)):
            current, previous = self.chain[i], self.chain[i - 1]
            if current.hash != current.calculate_hash() or current.previous_hash != previous.hash:
                return False
        return True

chain = Blockchain()
chain.add_transaction("d", 100)
chain.add_transaction("w", 200)

for b in chain.chain:
    print(b.index, b.operation, b.balance, b.hash)
    
print("Blockchain valid:", chain.is_valid())
```

```solidity
contract Palindrome {
    function isPalindrome(uint256 number) public pure returns (bool) {
        uint256 original = number;
        uint256 reversed = 0;
        
        while (number > 0) {
            reversed = reversed * 10 + number % 10;
            number /= 10;
        }
        return original == reversed;
    }
}
```

# Practical 4F – Mean, Median and Mode Smart Contract (Remix, Solidity)

## 1. Code

**File: Statistics.sol**
```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.21;
 
contract Statistics {
    function mean(uint256[] memory numbers) public pure returns (uint256) {
        require(numbers.length > 0, "Empty array");
        uint256 sum = 0;
        for (uint256 i = 0; i < numbers.length; i++) {
            sum += numbers[i];
        }
        return sum / numbers.length;
    }
 
    function median(uint256[] memory numbers) public pure returns (uint256) {
        require(numbers.length > 0, "Empty array");
        for (uint256 i = 0; i < numbers.length; i++) {
            for (uint256 j = i + 1; j < numbers.length; j++) {
                if (numbers[j] < numbers[i]) {
                    uint256 temp = numbers[i];
                    numbers[i] = numbers[j];
                    numbers[j] = temp;
                }
            }
        }
        uint256 mid = numbers.length / 2;
        if (numbers.length % 2 == 1) return numbers[mid];
        return (numbers[mid - 1] + numbers[mid]) / 2;
    }
 
    function mode(uint256[] memory numbers) public pure returns (uint256) {
        require(numbers.length > 0, "Empty array");
        uint256 bestValue = numbers[0];
        uint256 bestCount = 0;
        for (uint256 i = 0; i < numbers.length; i++) {
            uint256 count = 0;
            for (uint256 j = 0; j < numbers.length; j++) {
                if (numbers[j] == numbers[i]) count++;
            }
            if (count > bestCount) {
                bestCount = count;
                bestValue = numbers[i];
            }
        }
        return bestValue;
    }
}
```

# Practical 5B – Student Test Scores and Grades Using Struct

## 1. Code

**File: StudentGrades.sol**
```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.21;
 
contract StudentGrades {
    struct Student {
        string name;
        uint256 rollNo;
        uint256 test1;
        uint256 test2;
        uint256 test3;
    }
 
    mapping(uint256 => Student) public students;
 
    function setStudent(
        uint256 rollNo, string memory name,
        uint256 test1, uint256 test2, uint256 test3
    ) public {
        require(test1 <= 100 && test2 <= 100 && test3 <= 100, "Marks must be 0-100");
        students[rollNo] = Student(name, rollNo, test1, test2, test3);
    }
 
    function getTotal(uint256 rollNo) public view returns (uint256) {
        Student memory s = students[rollNo];
        return s.test1 + s.test2 + s.test3;
    }
 
    function getAverage(uint256 rollNo) public view returns (uint256) {
        return getTotal(rollNo) / 3;
    }
 
    function getGrade(uint256 rollNo) public view returns (string memory) {
        uint256 average = getAverage(rollNo);
        if (average >= 90) return "A+";
        if (average >= 80) return "A";
        if (average >= 70) return "B";
        if (average >= 60) return "C";
        return "Fail";
    }
}
```

```solidity
        require(msg.value > 0, "Deposit must be greater than zero");
        accounts[msg.sender].balance += msg.value;
        accounts[msg.sender].transactions.push(Transaction("Deposit", msg.value));
    }
 
    function withdraw(uint256 amount) public {
        require(accounts[msg.sender].balance >= amount, "Insufficient balance");
        accounts[msg.sender].balance -= amount;
        accounts[msg.sender].transactions.push(Transaction("Withdrawal", amount));
        payable(msg.sender).transfer(amount);
    }
 
    function getBalance() public view returns (uint256) {
        return accounts[msg.sender].balance;
    }
 
    function getTransactionCount() public view returns (uint256) {
        return accounts[msg.sender].transactions.length;
    }
}
```

```solidity
        require(correctOption < 4, "Invalid correct option");
        questions.push(Question(text, options, correctOption));
    }

        require(index < questions.length, "Invalid question");
        require(selectedOption < 4, "Invalid option");

        if (isCorrect) correct++;
        else wrong++;
        
        emit AnswerResult(questions[index].text, questions[index].options[selectedOption], isCorrect);
    }
    
    function getQuestion(uint256 index) public view returns (string memory, string[4] memory, uint8) {
        require(index < questions.length, "Invalid question");
        Question storage q = questions[index];
        return (q.text, q.options, q.correctOption);
    }
    
    function getScore() public view returns (uint256, uint256) {
        return (correct, wrong);
```

```solidity
    uint256 public nextAppointmentId = 1;
 
    struct Appointment {
        address patient;
        string doctorName;
        uint256 timeSlot;
        bool cancelled;
    }
 
    mapping(uint256 => Appointment) public appointments;
 
    constructor() { owner = msg.sender; }
 
    function bookAppointment(string memory doctorName, uint256 timeSlot)
        public payable returns (uint256)
    {
        require(msg.value == appointmentFee, "Incorrect fee");
        uint256 id = nextAppointmentId++;
        appointments[id] = Appointment(msg.sender, doctorName, timeSlot, false);
        return id;
    }
 
    function cancelAppointment(uint256 appointmentId) public {
        Appointment storage a = appointments[appointmentId];
        require(a.patient == msg.sender, "Not your appointment");
        require(!a.cancelled, "Already cancelled");
        
        a.cancelled = true;
        payable(msg.sender).transfer(appointmentFee);
    }
 
    function setAppointmentFee(uint256 newFee) public {
        require(msg.sender == owner, "Only owner");
        appointmentFee = newFee;
    }
}
```
